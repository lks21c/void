#### 커서 위치 기반 수집

```typescript
class ContextGatheringService {
  private visitedIntervals: Map<string, Set<string>> = new Map();
  private cachedSnippets: CodeSnippet[] = [];

  async updateCache(model: ITextModel, pos: Position): Promise<void> {
    const uri = model.uri.toString();

    // 1. 주변 코드 수집 (앞뒤 50줄)
    const surroundingLines = this.getSurroundingLines(model, pos, 50);

    // 2. 현재 함수/클래스 범위 수집
    const containingScope = await this.getContainingScope(model, pos);

    // 3. 심볼 정의 수집
    const symbols = await this.getSymbolsAtPosition(model, pos);

    // 4. 중복 방지
    const intervalKey = `${pos.lineNumber}-${pos.column}`;
    if (!this.hasVisited(uri, intervalKey)) {
      this.cachedSnippets.push(...this.createSnippets(
        surroundingLines,
        containingScope,
        symbols
      ));
      this.markVisited(uri, intervalKey);
    }
  }

  private hasVisited(uri: string, interval: string): boolean {
    const intervals = this.visitedIntervals.get(uri);
    return intervals?.has(interval) ?? false;
  }

  private markVisited(uri: string, interval: string): void {
    if (!this.visitedIntervals.has(uri)) {
      this.visitedIntervals.set(uri, new Set());
    }
    this.visitedIntervals.get(uri)!.add(interval);
  }
}
```

```typescript
  MAX_DIRECTORY_CHARS: 20_000,      // 디렉토리 트리 최대 문자

function optimizeContext(context: ContextResult): ContextResult {
  // 1. 중복 제거
  const uniqueSnippets = deduplicateSnippets(context.snippets);

  // 2. 우선순위 정렬
  const prioritized = sortByRelevance(uniqueSnippets);

  // 3. 크기 제한 적용
  const truncated = truncateToLimit(prioritized, CONTEXT_LIMITS.MAX_TOTAL_CONTEXT);

  return {
    ...context,
    snippets: truncated
  };
}

function deduplicateSnippets(snippets: CodeSnippet[]): CodeSnippet[] {
  const seen = new Set<string>();
  return snippets.filter(snippet => {
    const key = `${snippet.uri}:${snippet.range.startLine}-${snippet.range.endLine}`;
    if (seen.has(key)) return false;
    seen.add(key);
    return true;
  });
}
```

### 3.2 Fast Apply 구현

```typescript
interface EditFileTool {
  path: string;
  searchReplaceBlocks: SearchReplaceBlock[];
}

interface SearchReplaceBlock {
  search: string;
  replace: string;
}

class FastApplyHandler {
  async apply(edit: EditFileTool): Promise<ApplyResult> {
    const document = await this.documentService.open(edit.path);
    let content = document.getText();
    const changes: AppliedChange[] = [];

    // 1. 각 블록에 대해 검색 및 위치 기록
    for (const block of edit.searchReplaceBlocks) {
      const index = content.indexOf(block.search);

      if (index === -1) {
        // 유사 매칭 시도
        const fuzzyMatch = this.findFuzzyMatch(content, block.search);
        if (fuzzyMatch) {
          changes.push({
            offset: fuzzyMatch.index,
            length: fuzzyMatch.length,
            oldText: fuzzyMatch.text,
            newText: block.replace,
            confidence: fuzzyMatch.confidence
          });
        } else {
          throw new EditError(
            `검색 문자열을 찾을 수 없습니다: "${block.search.slice(0, 100)}..."`
          );
        }
      } else {
        changes.push({
          offset: index,
          length: block.search.length,
          oldText: block.search,
          newText: block.replace,
          confidence: 1.0
        });
      }
    }

    // 2. 역순으로 정렬 (뒤에서부터 적용하여 오프셋 유지)
    changes.sort((a, b) => b.offset - a.offset);

    // 3. 변경 적용
    for (const change of changes) {
      content =
        content.slice(0, change.offset) +
        change.newText +
        content.slice(change.offset + change.length);
    }

    // 4. 파일 저장
    await this.documentService.save(edit.path, content);

    return {
      success: true,
      changesApplied: changes.length,
      changes
    };
  }

  private findFuzzyMatch(
    content: string,
    search: string
  ): FuzzyMatchResult | null {
    // 공백 정규화 후 매칭
    const normalizedSearch = search.replace(/\s+/g, '\\s+');
    const regex = new RegExp(normalizedSearch);
    const match = content.match(regex);

    if (match) {
      return {
        index: match.index!,
        length: match[0].length,
        text: match[0],
        confidence: 0.9
      };
    }

    return null;
  }
}
```

### 3.4 Fast Apply vs Slow Apply 선택 기준

```typescript
function selectApplyStrategy(
  originalContent: string,
  proposedChanges: ProposedChange[]
): 'fast' | 'slow' {
  // 1. 변경 비율 계산
  const totalChangeSize = proposedChanges.reduce(
    (sum, change) => sum + Math.abs(change.newText.length - change.oldText.length),
    0
  );
  const changeRatio = totalChangeSize / originalContent.length;

  // 2. 50% 이상 변경이면 Slow Apply
  if (changeRatio > 0.5) {
    return 'slow';
  }

  // 3. 구조적 변경 감지 (함수/클래스 추가/삭제)
  const hasStructuralChanges = proposedChanges.some(change =>
    /^(function|class|interface|type)\s/.test(change.newText) ||
    /^(function|class|interface|type)\s/.test(change.oldText)
  );

  if (hasStructuralChanges && proposedChanges.length > 5) {
    return 'slow';
  }

  // 4. 기본은 Fast Apply
  return 'fast';
}
```

### 4.1 DiffZone 구조

```typescript
interface DiffZone {

  | 'pending'      // 생성됨, 대기 중
  | 'streaming'    // LLM 응답 스트리밍 중
  | 'ready'        // 적용 준비 완료
  | 'applied'      // 적용 완료
  | 'cancelled'    // 취소됨
  | 'error';       // 오류 발생
```

### 4.2 DiffZone 관리자

```typescript
class DiffZoneManager extends Disposable {
  private zones = new Map<string, DiffZone>();
  private decorations = new Map<string, string[]>();

  private readonly _onZoneCreated = this._register(new Emitter<DiffZone>());
  readonly onZoneCreated = this._onZoneCreated.event;

  private readonly _onZoneUpdated = this._register(new Emitter<DiffZone>());
  readonly onZoneUpdated = this._onZoneUpdated.event;

  createZone(uri: URI, range: IRange): DiffZone {
    const document = this.documentService.get(uri);
    const originalContent = document.getValueInRange(range);

    const zone: DiffZone = {
      id: generateId(),
      uri,
      originalRange: range,
      originalContent,
      modifiedContent: '',
      status: 'pending',
      createdAt: Date.now()
    };

    this.zones.set(zone.id, zone);
    this.createDecorations(zone);
    this._onZoneCreated.fire(zone);

    return zone;
  }

  updateStreaming(zoneId: string, content: string): void {
    const zone = this.zones.get(zoneId);
    if (!zone) return;

    zone.modifiedContent = content;
    zone.status = 'streaming';

    this.updateDecorations(zone);
    this._onZoneUpdated.fire(zone);
  }

  finishStreaming(zoneId: string): void {
    const zone = this.zones.get(zoneId);
    if (!zone) return;

    zone.status = 'ready';
    this.updateDecorations(zone);
    this._onZoneUpdated.fire(zone);
  }

  async applyZone(zoneId: string): Promise<void> {
    const zone = this.zones.get(zoneId);
    if (!zone || zone.status !== 'ready') return;

    // 체크포인트 생성
    const checkpoint = this.checkpointService.create(zone.uri);

    try {
      // 편집 적용
      await this.documentService.edit(zone.uri, [{
        range: zone.originalRange,
        text: zone.modifiedContent
      }]);

      zone.status = 'applied';
      this.clearDecorations(zone);
      this._onZoneUpdated.fire(zone);
    } catch (error) {
      zone.status = 'error';
      await this.checkpointService.rollback(checkpoint.id);
      throw error;
    }
  }

  cancelZone(zoneId: string): void {
    const zone = this.zones.get(zoneId);
    if (!zone) return;

    zone.status = 'cancelled';
    this.clearDecorations(zone);
    this.zones.delete(zoneId);
  }
}
```

### 4.3 Diff 시각화

```typescript
class DiffDecorationProvider {
  private readonly deletedLineClass = 'diff-deleted-line';
  private readonly insertedLineClass = 'diff-inserted-line';
  private readonly modifiedLineClass = 'diff-modified-line';

  createDecorations(zone: DiffZone): EditorDecoration[] {
    const diff = computeLineDiff(zone.originalContent, zone.modifiedContent);
    const decorations: EditorDecoration[] = [];

    for (const change of diff.changes) {
      switch (change.type) {
        case 'delete':
          decorations.push({
            range: change.range,
            options: {
              className: this.deletedLineClass,
              isWholeLine: true,
              linesDecorationsClassName: 'diff-gutter-deleted'
            }
          });
          break;

        case 'insert':
          decorations.push({
            range: change.range,
            options: {
              className: this.insertedLineClass,
              isWholeLine: true,
              linesDecorationsClassName: 'diff-gutter-inserted'
            }
          });
          break;

        case 'modify':
          decorations.push({
            range: change.range,
            options: {
              className: this.modifiedLineClass,
              isWholeLine: true
            }
          });
          // 인라인 변경 하이라이트
          for (const inline of change.inlineChanges) {
            decorations.push({
              range: inline.range,
              options: {
                className: inline.type === 'delete'
                  ? 'diff-inline-deleted'
                  : 'diff-inline-inserted'
              }
            });
          }
          break;
      }
    }

    return decorations;
  }
}
```

### 5.2 체크포인트 서비스

```typescript
class CheckpointService extends Disposable {
  private readonly store: CheckpointStore = {
    entries: [],
    maxEntries: 100,
    maxAge: 24 * 60 * 60 * 1000 // 24시간
  };

  create(uri: URI, metadata?: CheckpointMetadata): CheckpointEntry {
    const document = this.documentService.get(uri);

    const entry: CheckpointEntry = {
      id: generateId(),
      uri,
      originalContent: document.getText(),
      timestamp: Date.now(),
      metadata
    };

    this.store.entries.push(entry);
    this.cleanup();

    return entry;
  }

  async rollback(checkpointId: string): Promise<void> {
    const entry = this.store.entries.find(e => e.id === checkpointId);
    if (!entry) {
      throw new Error(`Checkpoint not found: ${checkpointId}`);
    }

    await this.documentService.save(entry.uri, entry.originalContent);

    // 해당 체크포인트 이후의 모든 체크포인트 제거
    const index = this.store.entries.indexOf(entry);
    this.store.entries = this.store.entries.slice(0, index);
  }

  getHistory(uri: URI): CheckpointEntry[] {
    return this.store.entries
      .filter(e => e.uri.toString() === uri.toString())
      .sort((a, b) => b.timestamp - a.timestamp);
  }

  private cleanup(): void {
    const now = Date.now();

    // 오래된 항목 제거
    this.store.entries = this.store.entries.filter(
      e => now - e.timestamp < this.store.maxAge
    );

    // 최대 개수 초과 시 오래된 것부터 제거
    if (this.store.entries.length > this.store.maxEntries) {
      this.store.entries = this.store.entries.slice(-this.store.maxEntries);
    }
  }
}
```

```typescript
class StagingService extends Disposable {
  private state: StagingState = {
    items: [],
    files: [],
    folders: []
  };

  private readonly _onStateChanged = this._register(new Emitter<StagingState>());
  readonly onStateChanged = this._onStateChanged.event;

  addSelection(uri: URI, range: IRange): void {
    const document = this.documentService.get(uri);
    const content = document.getValueInRange(range);

    const item: StagingSelectionItem = {
      id: generateId(),
      uri,
      range,
      content,
      type: 'code',
      addedAt: Date.now()
    };

    this.state.items.push(item);
    this._onStateChanged.fire(this.state);
  }

  addFile(uri: URI): void {
    if (!this.state.files.some(f => f.toString() === uri.toString())) {
      this.state.files.push(uri);
      this._onStateChanged.fire(this.state);
    }
  }

  removeItem(id: string): void {
    this.state.items = this.state.items.filter(item => item.id !== id);
    this._onStateChanged.fire(this.state);
  }

  clear(): void {
    this.state = { items: [], files: [], folders: [] };
    this._onStateChanged.fire(this.state);
  }

  // LLM 메시지용 컨텍스트 생성
  toContext(): ContextAttachment[] {
    const attachments: ContextAttachment[] = [];

    // 코드 선택 항목
    for (const item of this.state.items) {
      attachments.push({
        type: 'code',
        uri: item.uri,
        range: item.range,
        content: item.content
      });
    }

    // 전체 파일
    for (const uri of this.state.files) {
      const document = this.documentService.get(uri);
      attachments.push({
        type: 'file',
        uri,
        content: document.getText()
      });
    }

    return attachments;
  }
}
```

## 7. 스트리밍 코드 생성

### 7.1 스트리밍 응답 처리

```typescript
interface StreamingCodeHandler {
  onStart(requestId: string): void;
  onChunk(chunk: string): void;
  onToolUse(toolCall: ToolCall): void;
  onComplete(result: CompletionResult): void;
  onError(error: Error): void;
}

class CodeGenerationStreamHandler implements StreamingCodeHandler {
  private buffer = '';
  private currentDiffZone: DiffZone | null = null;

  constructor(
    private readonly diffZoneManager: DiffZoneManager,
    private readonly toolExecutor: ToolExecutor
  ) {}

  onStart(requestId: string): void {
    this.buffer = '';
    this.currentDiffZone = null;
  }

  onChunk(chunk: string): void {
    this.buffer += chunk;

    // Diff 마커 감지
    if (this.detectDiffStart(this.buffer)) {
      if (!this.currentDiffZone) {
        // 새 DiffZone 생성
        const range = this.extractTargetRange(this.buffer);
        this.currentDiffZone = this.diffZoneManager.createZone(
          this.currentUri,
          range
        );
      }

      // 스트리밍 업데이트
      const modifiedContent = this.extractModifiedContent(this.buffer);
      this.diffZoneManager.updateStreaming(
        this.currentDiffZone.id,
        modifiedContent
      );
    }
  }

  onToolUse(toolCall: ToolCall): void {
    // 도구 호출 처리
    this.toolExecutor.execute(toolCall).then(result => {
      // 결과를 다음 메시지에 포함
    });
  }

  onComplete(result: CompletionResult): void {
    if (this.currentDiffZone) {
      this.diffZoneManager.finishStreaming(this.currentDiffZone.id);
    }
  }

  private detectDiffStart(content: string): boolean {
    return content.includes('<<<<<<< ORIGINAL');
  }

  private extractModifiedContent(content: string): string {
    const match = content.match(/=======\n([\s\S]*?)(?:>>>>>>> UPDATED|$)/);
    return match ? match[1] : '';
  }
}
```

```typescript
interface ParsedDiff {
  original: string;
  modified: string;
  isComplete: boolean;
}

function parseDiffMarkers(content: string): ParsedDiff[] {
  const diffs: ParsedDiff[] = [];
  const pattern = /<<<<<<< ORIGINAL\n([\s\S]*?)\n=======\n([\s\S]*?)\n>>>>>>> UPDATED/g;

  let match;
  while ((match = pattern.exec(content)) !== null) {
    diffs.push({
      original: match[1],
      modified: match[2],
      isComplete: true
    });
  }

  // 불완전한 diff 감지 (스트리밍 중)
  const incompleteMatch = content.match(/<<<<<<< ORIGINAL\n([\s\S]*?)\n=======\n([\s\S]*)$/);
  if (incompleteMatch && !incompleteMatch[2].includes('>>>>>>> UPDATED')) {
    diffs.push({
      original: incompleteMatch[1],
      modified: incompleteMatch[2],
      isComplete: false
    });
  }

  return diffs;
}
```

```typescript
interface PreApplyValidation {

interface ValidationResult {
  valid: boolean;
  errors: ValidationError[];
  warnings: ValidationWarning[];
}

class EditValidator implements PreApplyValidation {
  validate(edit: CodeEdit): ValidationResult {
    const errors: ValidationError[] = [];
    const warnings: ValidationWarning[] = [];

    // 1. 파일 존재 확인
    if (!this.fileService.exists(edit.path)) {
      errors.push({
        code: 'FILE_NOT_FOUND',
        message: `파일을 찾을 수 없습니다: ${edit.path}`
      });
    }

    // 2. 검색 문자열 존재 확인 (Fast Apply)
    if (edit.type === 'fast' && edit.searchReplaceBlocks) {
      const content = this.fileService.read(edit.path);
      for (const block of edit.searchReplaceBlocks) {
        if (!content.includes(block.search)) {
          errors.push({
            code: 'SEARCH_NOT_FOUND',
            message: `검색 문자열을 찾을 수 없습니다`,
            details: { search: block.search.slice(0, 100) }
          });
        }
      }
    }

    // 3. 구문 검증 (변경 후 코드)
    if (edit.type === 'slow' && edit.content) {
      const syntaxErrors = this.checkSyntax(edit.path, edit.content);
      errors.push(...syntaxErrors);
    }

    // 4. 변경 규모 경고
    if (edit.type === 'fast' && edit.searchReplaceBlocks.length > 10) {
      warnings.push({
        code: 'LARGE_CHANGE',
        message: '많은 수의 변경이 있습니다. Slow Apply를 고려해보세요.'
      });
    }

    return {
      valid: errors.length === 0,
      errors,
      warnings
    };
  }
}
```

```typescript
      });

      const related = await this.gatherRelatedCode(snippets);
      snippets.push(...related);
    }

    // 4. 최적화 및 반환
    return {
      snippets: this.optimizeSnippets(snippets),
      totalTokens: this.estimateTokens(snippets)
    };
  }

      const imports = this.parseImports(snippet.content);

      }
    }
```

```typescript
// hdsp-agent용 코드 변환 서비스
class AgentCodeTransformService {
  constructor(
    private readonly checkpointService: CheckpointService,
    private readonly validationService: ValidationService
  ) {}

  async applyTransform(transform: CodeTransform): Promise<TransformResult> {
    // 1. 사전 검증
    const validation = await this.validationService.validate(transform);
    if (!validation.valid) {
      return { success: false, errors: validation.errors };
    }

    // 2. 체크포인트 생성
    const checkpoints = await this.createCheckpoints(transform.files);

    try {
      // 3. 전략 선택 및 적용
      const strategy = this.selectStrategy(transform);
      const results = await strategy.apply(transform);

      // 4. 사후 검증
      const postValidation = await this.validateResults(results);
      if (!postValidation.valid) {
        await this.rollbackAll(checkpoints);
        return { success: false, errors: postValidation.errors };
      }

      return { success: true, results };
    } catch (error) {
      // 5. 오류 시 롤백
      await this.rollbackAll(checkpoints);
      throw error;
    }
  }

  private selectStrategy(transform: CodeTransform): TransformStrategy {
    // 변경 규모에 따른 전략 선택
    const totalChanges = transform.edits.length;
    const avgChangeSize = this.calculateAverageChangeSize(transform.edits);

    if (totalChanges <= 3 && avgChangeSize < 100) {
      return new FastApplyStrategy();
    } else {
      return new SlowApplyStrategy();
    }
  }
}
```

```typescript
class AgentDiffService {
  showDiff(original: string, modified: string): DiffPresentation {

    return {
      // 터미널 출력용
      terminal: this.formatForTerminal(diff),

      // 구조화된 데이터
      structured: {
        additions: diff.additions,
        deletions: diff.deletions,
        modifications: diff.modifications,
        unchanged: diff.unchanged
      },

      // 통계
      stats: {
        linesAdded: diff.additions.length,
        linesDeleted: diff.deletions.length,
        linesModified: diff.modifications.length
      }
    };
  }

  private formatForTerminal(diff: Diff): string {
    const lines: string[] = [];

    for (const change of diff.changes) {
      switch (change.type) {
        case 'add':
          lines.push(`+ ${change.content}`);
          break;
        case 'delete':
          lines.push(`- ${change.content}`);
          break;
        case 'context':
          lines.push(`  ${change.content}`);
          break;
      }
    }

    return lines.join('\n');
  }
}
```
