## 2. 에러 핸들러 체인

### 2.1 전역 에러 핸들러

```typescript
// 예상치 못한 에러 핸들러 설정
function setUnexpectedErrorHandler(handler: (error: Error) => void): void {
  globalErrorHandler = handler;
}

// 에러 핸들러 호출
function onUnexpectedError(error: Error): void {
  // 취소 에러는 무시
  if (isCancellationError(error)) {
    return;
  }

  // 외부 에러는 경고만
  if (isExternalError(error)) {
    console.warn('External error:', error);
    return;
  }

  // 전역 핸들러 호출
  globalErrorHandler(error);
}
```

### 2.2 다층 에러 핸들링

```typescript
class ErrorHandlerChain {
  private handlers: ErrorHandler[] = [];

  register(handler: ErrorHandler, priority: number): IDisposable {
    this.handlers.push({ handler, priority });
    this.handlers.sort((a, b) => b.priority - a.priority);

    return {
      dispose: () => {
        const index = this.handlers.findIndex(h => h.handler === handler);
        if (index >= 0) {
          this.handlers.splice(index, 1);
        }
      }
    };
  }

  handle(error: Error): boolean {
    for (const { handler } of this.handlers) {
      if (handler.canHandle(error)) {
        handler.handle(error);
        return true;
      }
    }
    return false;
  }
}

// 사용 예시
const chain = new ErrorHandlerChain();

// 네트워크 에러 핸들러 (우선순위 높음)
chain.register({
  canHandle: (e) => e instanceof NetworkError,
  handle: (e) => showNetworkErrorDialog(e)
}, 100);

// 일반 에러 핸들러 (우선순위 낮음)
chain.register({
  canHandle: () => true,
  handle: (e) => logError(e)
}, 0);
```

```typescript
  constructor() {

  constructor(

function isCancellationError(error: Error): boolean {
  return error instanceof CancellationError ||

  return error instanceof NetworkError ||
         error.name === 'NetworkError';
}
```

## 3. 취소 토큰 (CancellationToken)

### 3.1 인터페이스 정의

```typescript
interface CancellationToken {
  readonly isCancellationRequested: boolean;
  readonly onCancellationRequested: Event<any>;
}

const CancellationToken = {
  None: Object.freeze<CancellationToken>({
    isCancellationRequested: false,
    onCancellationRequested: Event.None
  }),

  Cancelled: Object.freeze<CancellationToken>({
    isCancellationRequested: true,
    onCancellationRequested: Event.None
  })
};
```

### 3.2 CancellationTokenSource

```typescript
class CancellationTokenSource {
  private _token?: CancellationToken;
  private _parentListener?: IDisposable;

  get token(): CancellationToken {
    if (!this._token) {
      this._token = new MutableToken();
    }
    return this._token;
  }

  cancel(): void {
    if (!this._token) {
      this._token = CancellationToken.Cancelled;
    } else if (this._token instanceof MutableToken) {
      this._token.cancel();
    }
  }

  dispose(cancel: boolean = false): void {
    if (cancel) {
      this.cancel();
    }
    this._parentListener?.dispose();
    if (this._token instanceof MutableToken) {
      this._token.dispose();
    }
  }
}

class MutableToken implements CancellationToken {
  private _isCancelled = false;
  private _emitter?: Emitter<any>;

  get isCancellationRequested(): boolean {
    return this._isCancelled;
  }

  get onCancellationRequested(): Event<any> {
    if (!this._emitter) {
      this._emitter = new Emitter<any>();
    }
    return this._emitter.event;
  }

  cancel(): void {
    if (!this._isCancelled) {
      this._isCancelled = true;
      this._emitter?.fire(undefined);
    }
  }

  dispose(): void {
    this._emitter?.dispose();
  }
}
```

### 3.3 취소 처리 패턴

```typescript
async function fetchWithCancellation(
  url: string,
  token: CancellationToken
): Promise<Response> {
  // 이미 취소됨
  if (token.isCancellationRequested) {
    throw new CancellationError();
  }

  const controller = new AbortController();

  // 취소 시 abort
  const listener = token.onCancellationRequested(() => {
    controller.abort();
  });

  try {
    const response = await fetch(url, { signal: controller.signal });
    return response;
  } catch (error) {
    if (error.name === 'AbortError') {
      throw new CancellationError();
    }
    throw error;
  } finally {
    listener.dispose();
  }
}
```

## 4. 재시도 패턴

### 4.1 지수 백오프

```typescript
interface RetryOptions {
  maxRetries: number;
  initialDelay: number;
  maxDelay: number;
  factor: number;
  retryOn?: (error: Error) => boolean;
}

async function withRetry<T>(
  fn: () => Promise<T>,
  options: RetryOptions
): Promise<T> {
  const {
    maxRetries,
    initialDelay,
    maxDelay,
    factor,
    retryOn = () => true
  } = options;

  let lastError: Error;
  let delay = initialDelay;

  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    try {
      return await fn();
    } catch (error) {
      lastError = error;

      // 재시도 불가능한 에러
      if (!retryOn(error)) {
        throw error;
      }

      // 마지막 시도였음
      if (attempt === maxRetries) {
        throw error;
      }

      // 대기 후 재시도
      await sleep(delay);
      delay = Math.min(delay * factor, maxDelay);
    }
  }

  throw lastError!;
}

// 사용 예시
const result = await withRetry(
  () => fetchData(url),
  {
    maxRetries: 3,
    initialDelay: 1000,
    maxDelay: 10000,
    factor: 2,
    retryOn: (e) => isNetworkError(e) && e.retryable
  }
);
```

### 4.2 재시도 가능한 에러 판별

```typescript
function isRetryableError(error: Error): boolean {
  // 취소는 재시도 불가
  if (isCancellationError(error)) {
    return false;
  }

  // 네트워크 에러
  if (error instanceof NetworkError) {
    // 4xx는 재시도 불가 (클라이언트 오류)
    if (error.statusCode && error.statusCode >= 400 && error.statusCode < 500) {
      return false;
    }
    return error.retryable;
  }

  // 타임아웃
  if (error.message.includes('timeout')) {
    return true;
  }

  // 일시적 에러
  if (error.message.includes('temporarily unavailable')) {
    return true;
  }

  return false;
}
```

## 5. Disposable 패턴

### 5.1 기본 인터페이스

```typescript
interface IDisposable {
  dispose(): void;
}

// Disposable 체크
function isDisposable(obj: any): obj is IDisposable {
  return typeof obj?.dispose === 'function';
}
```

### 5.2 Disposable 클래스

```typescript
class Disposable implements IDisposable {
  static readonly None = Object.freeze<IDisposable>({ dispose() {} });

  protected readonly _store = new DisposableStore();

  protected _register<T extends IDisposable>(t: T): T {
    if ((t as unknown) === this) {
      throw new Error('Cannot register self');
    }
    return this._store.add(t);
  }

  dispose(): void {
    this._store.dispose();
  }
}
```

### 5.3 DisposableStore

```typescript
class DisposableStore implements IDisposable {
  private _toDispose = new Set<IDisposable>();
  private _isDisposed = false;

  add<T extends IDisposable>(t: T): T {
    if (this._isDisposed) {
      console.warn('Adding to disposed store');
      t.dispose();
    } else {
      this._toDispose.add(t);
    }
    return t;
  }

  delete(t: IDisposable): void {
    this._toDispose.delete(t);
  }

  dispose(): void {
    if (this._isDisposed) return;

    this._isDisposed = true;

    for (const disposable of this._toDispose) {
      try {
        disposable.dispose();
      } catch (e) {
        console.error('Error disposing', e);
      }
    }

    this._toDispose.clear();
  }
}
```

### 5.4 MutableDisposable

```typescript
class MutableDisposable<T extends IDisposable> implements IDisposable {
  private _value?: T;
  private _isDisposed = false;

  get value(): T | undefined {
    return this._isDisposed ? undefined : this._value;
  }

  set value(value: T | undefined) {
    if (this._isDisposed || value === this._value) {
      return;
    }

    this._value?.dispose();
    this._value = value;
  }

  clear(): void {
    this.value = undefined;
  }

  dispose(): void {
    this._isDisposed = true;
    this._value?.dispose();
    this._value = undefined;
  }
}
```

### 5.5 리소스 누수 추적

```typescript
class DisposableTracker {
  private static _enabled = false;
  private static _tracked = new Map<IDisposable, Error>();

  static setTracking(enabled: boolean): void {
    this._enabled = enabled;
  }

  static trackDisposable(d: IDisposable): void {
    if (!this._enabled) return;

    // 생성 위치 기록
    this._tracked.set(d, new Error('Disposable created here'));
  }

  static markAsDisposed(d: IDisposable): void {
    this._tracked.delete(d);
  }

  static reportLeaks(): void {
    if (this._tracked.size === 0) {
      console.log('No leaks detected');
      return;
    }

    console.error(`${this._tracked.size} leaks detected:`);
    for (const [_, stack] of this._tracked) {
      console.error(stack);
    }
  }
}
```

## 6. 에러 복구 전략

### 6.1 서킷 브레이커

```typescript
enum CircuitState {
  Closed,    // 정상 동작
  Open,      // 차단됨
  HalfOpen   // 테스트 중
}

class CircuitBreaker {
  private state = CircuitState.Closed;
  private failures = 0;
  private lastFailure = 0;

  constructor(
    private readonly threshold: number = 5,
    private readonly timeout: number = 30000
  ) {}

  async execute<T>(fn: () => Promise<T>): Promise<T> {
    if (this.state === CircuitState.Open) {
      if (Date.now() - this.lastFailure > this.timeout) {
        this.state = CircuitState.HalfOpen;
      } else {
        throw new Error('Circuit breaker is open');
      }
    }

    try {
      const result = await fn();
      this.onSuccess();
      return result;
    } catch (error) {
      this.onFailure();
      throw error;
    }
  }

  private onSuccess(): void {
    this.failures = 0;
    this.state = CircuitState.Closed;
  }

  private onFailure(): void {
    this.failures++;
    this.lastFailure = Date.now();

    if (this.failures >= this.threshold) {
      this.state = CircuitState.Open;
    }
  }
}
```

### 6.2 폴백 패턴

```typescript
async function withFallback<T>(
  primary: () => Promise<T>,
  fallback: () => Promise<T>,
  options?: { onFallback?: (error: Error) => void }
): Promise<T> {
  try {
    return await primary();
  } catch (error) {
    options?.onFallback?.(error);
    return await fallback();
  }
}

// 사용 예시
const result = await withFallback(
  () => fetchFromApi(url),
  () => getFromCache(cacheKey),
  { onFallback: (e) => console.warn('Using fallback:', e.message) }
);
```

```typescript
interface SerializedError {

function transformErrorForSerialization(error: Error): SerializedError {
  return {
    name: error.name,
    message: error.message,
    stack: error.stack,
    data: (error as any).data
  };
}

function deserializeError(serialized: SerializedError): Error {
  const error = new Error(serialized.message);
  error.name = serialized.name;
  error.stack = serialized.stack;
  (error as any).data = serialized.data;
  return error;
```

## 8. hdsp-agent 적용 가이드

### 8.1 에이전트 에러 핸들러

```typescript
class AgentErrorHandler {
  private readonly errorChain = new ErrorHandlerChain();

  constructor() {
    this.setupHandlers();
  }

  private setupHandlers(): void {
    // LLM 에러
    this.errorChain.register({
      canHandle: (e) => e instanceof LLMError,
      handle: (e) => this.handleLLMError(e as LLMError)
    }, 100);

    // 도구 실행 에러
    this.errorChain.register({
      canHandle: (e) => e instanceof ToolExecutionError,
      handle: (e) => this.handleToolError(e as ToolExecutionError)
    }, 90);

    // 네트워크 에러
    this.errorChain.register({
      canHandle: (e) => isNetworkError(e),
      handle: (e) => this.handleNetworkError(e)
    }, 80);
  }

  async handleError(error: Error): Promise<RecoveryAction> {
    if (!this.errorChain.handle(error)) {
      // 미처리 에러
      console.error('Unhandled error:', error);
      return RecoveryAction.Abort;
    }

    return this.determineRecoveryAction(error);
  }

  private determineRecoveryAction(error: Error): RecoveryAction {
    if (isCancellationError(error)) {
      return RecoveryAction.Cancel;
    }

    if (isRetryableError(error)) {
      return RecoveryAction.Retry;
    }

    return RecoveryAction.Abort;
  }
}
```

### 8.3 작업 취소

```typescript
class CancellableTask<T> {
  private cts = new CancellationTokenSource();
  private result: Promise<T>;

  constructor(task: (token: CancellationToken) => Promise<T>) {
    this.result = task(this.cts.token);
  }

  cancel(): void {
    this.cts.cancel();
  }

  async wait(): Promise<T> {
    return this.result;
  }

  dispose(): void {
    this.cts.dispose(true);
  }
}

// 사용 예시
const task = new CancellableTask(async (token) => {
  const result = await longRunningOperation(token);
  return result;
});

// 필요시 취소
setTimeout(() => task.cancel(), 5000);
```
