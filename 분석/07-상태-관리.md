## 3. Observable 패턴

### 3.1 Observable 인터페이스

```typescript
interface IObservable<T> {
  get(): T;
  subscribe(observer: IObserver<T>): IDisposable;
}

interface IObserver<T> {
  handleChange(value: T): void;
}

interface ISettableObservable<T> extends IObservable<T> {
  set(value: T): void;
}
```

### 3.2 Observable 구현

```typescript
class Observable<T> implements ISettableObservable<T> {
  private _value: T;
  private readonly _observers = new Set<IObserver<T>>();

  constructor(initialValue: T) {
    this._value = initialValue;
  }

  get(): T {
    return this._value;
  }

  set(value: T): void {
    if (this._value !== value) {
      this._value = value;
      this.notify();
    }
  }

  subscribe(observer: IObserver<T>): IDisposable {
    this._observers.add(observer);
    return {
      dispose: () => this._observers.delete(observer)
    };
  }

  private notify(): void {
    for (const observer of this._observers) {
      observer.handleChange(this._value);
    }
  }
}
```

```typescript
    }
  }
```

### 4.2 설정 검증

```typescript
class VoidSettingsValidator {
  validate(settings: SettingsOfProvider): ValidationResult {
    const errors: string[] = [];

    for (const [provider, config] of Object.entries(settings)) {
      // API 키 검증
      if (this.requiresApiKey(provider) && !config.apiKey) {
        errors.push(`${provider}: API 키가 필요합니다`);
      }

      // URL 형식 검증
      if (config.baseUrl && !this.isValidUrl(config.baseUrl)) {
        errors.push(`${provider}: 올바른 URL이 아닙니다`);
      }

      // 모델 검증
      if (config.models) {
        const validModels = this.getValidModels(provider);
        for (const model of config.models) {
          if (!validModels.includes(model)) {
            errors.push(`${provider}: 알 수 없는 모델 - ${model}`);
          }
        }
      }
    }

    return {
      valid: errors.length === 0,
      errors
    };
  }
}
```

```typescript
class ChatThreadStorage {
  private readonly storageKey = 'void.chatThreads';

  async save(threads: IChatThread[]): Promise<void> {
    const serialized = threads.map(thread => ({
      ...thread,
      messages: thread.messages.map(msg => this.serializeMessage(msg))
    }));

    await this.storage.store(this.storageKey, JSON.stringify(serialized));
  }

  async load(): Promise<IChatThread[]> {
    const data = await this.storage.get(this.storageKey);
    if (!data) return [];

    const parsed = JSON.parse(data);
    return parsed.map(thread => ({
      ...thread,
      messages: thread.messages.map(msg => this.deserializeMessage(msg))
    }));
  }

  private serializeMessage(msg: IChatMessage): SerializedMessage {
    return {
      ...msg,
      content: msg.content.map(c => {
        if (c.type === 'image') {
          return { ...c, data: c.data.toString('base64') };
        }
        return c;
      })
    };
  }
}
```

```typescript
class SessionManager {
  private currentSession: ISession | null = null;

  async startSession(): Promise<ISession> {
    // 이전 세션 종료
    if (this.currentSession) {
      await this.endSession();
    }

    // 새 세션 생성
    this.currentSession = {
      id: generateId(),
      startTime: Date.now(),
      threads: [],
      activeThread: '',
      settings: this.getDefaultSettings()
    };

    // 저장
    await this.saveSession();

    return this.currentSession;
  }

  async endSession(): Promise<void> {
    if (!this.currentSession) return;

    this.currentSession.endTime = Date.now();
    await this.saveSession();
    this.currentSession = null;
  }

  async resumeSession(sessionId: string): Promise<ISession | null> {
    const session = await this.loadSession(sessionId);
    if (!session) return null;

    this.currentSession = session;
    return session;
  }
}
```

```typescript
class AutoCheckpointService {
  private readonly triggers = [
    'tool_execution',   // 도구 실행 전
    'file_edit',        // 파일 편집 전
    'batch_operation',  // 배치 작업 전
  ];

  async maybeCreateCheckpoint(
    trigger: string,
    uri: URI
  ): Promise<CheckpointEntry | null> {
    if (!this.triggers.includes(trigger)) {
      return null;
    }

    // 중복 방지 (최근 1분 내 동일 파일)
    const recent = await this.store.list({
      uri,
      afterTime: Date.now() - 60000
    });

    if (recent.length > 0) {
      return recent[0];
    }

    // 체크포인트 생성
    const content = await this.fileService.readFile(uri);
    const entry: CheckpointEntry = {
      id: generateId(),
      uri,
      originalContent: content.toString(),
      timestamp: Date.now(),
      metadata: { trigger }
    };

    await this.store.create(entry);
    return entry;
  }
}
```

### 8.1 더티 상태 추적

```typescript
interface IFileState {
  uri: URI;
  isDirty: boolean;
  lastSaved: number;
  lastModified: number;
  version: number;
}

class FileStateTracker {
  private states = new Map<string, IFileState>();

  private readonly _onStateChanged = new Emitter<IFileState>();
  readonly onStateChanged = this._onStateChanged.event;

  markDirty(uri: URI): void {
    const state = this.getOrCreate(uri);
    state.isDirty = true;
    state.lastModified = Date.now();
    state.version++;
    this._onStateChanged.fire(state);
  }

  markSaved(uri: URI): void {
    const state = this.getOrCreate(uri);
    state.isDirty = false;
    state.lastSaved = Date.now();
    this._onStateChanged.fire(state);
  }

  getDirtyFiles(): URI[] {
    return Array.from(this.states.values())
      .filter(s => s.isDirty)
      .map(s => s.uri);
  }
}
```

### 8.2 충돌 감지

```typescript
class ConflictDetector {
  async checkConflict(uri: URI, baseVersion: number): Promise<ConflictInfo | null> {
    const currentState = this.fileStateTracker.get(uri);

    if (currentState && currentState.version !== baseVersion) {
      return {
        uri,
        baseVersion,
        currentVersion: currentState.version,
        type: 'version_mismatch'
      };
    }

    // 외부 변경 감지
    const diskStat = await this.fileService.stat(uri);
    const lastKnownModified = this.getLastKnownModified(uri);

    if (diskStat.mtime > lastKnownModified) {
      return {
        uri,
        type: 'external_change',
        diskModified: diskStat.mtime
      };
    }

    return null;
  }
}
```

```typescript
// 계층적 설정 관리자
class AgentConfigManager {
  private configs: Map<ConfigLevel, AgentConfig> = new Map();

  constructor() {
    this.configs.set('default', this.getDefaultConfig());
    this.configs.set('user', new AgentConfig());
    this.configs.set('session', new AgentConfig());
  }

  get<T>(key: string): T {
    // session > user > default 순서로 탐색
    for (const level of ['session', 'user', 'default']) {
      const config = this.configs.get(level as ConfigLevel);
      if (config?.has(key)) {
        return config.get(key);
      }
    }
    throw new Error(`Config not found: ${key}`);
  }

  set(key: string, value: any, level: ConfigLevel = 'user'): void {
    this.configs.get(level)?.set(key, value);
    this._onConfigChanged.fire({ key, value, level });
  }
}
```

```typescript
// 에이전트 상태 관리
interface IAgentState {
  status: 'idle' | 'thinking' | 'executing' | 'waiting';
  currentTask: TaskInfo | null;
  history: TaskResult[];
  context: ContextState;
}

class AgentStateManager {
  private state: Observable<IAgentState>;

  constructor() {
    this.state = new Observable({
      status: 'idle',
      currentTask: null,
      history: [],
      context: {}
    });
  }

  transition(action: StateAction): void {
    const current = this.state.get();

    switch (action.type) {
      case 'START_TASK':
        this.state.set({
          ...current,
          status: 'thinking',
          currentTask: action.task
        });
        break;

      case 'EXECUTE_TOOL':
        this.state.set({
          ...current,
          status: 'executing'
        });
        break;

      case 'COMPLETE_TASK':
        this.state.set({
          ...current,
          status: 'idle',
          currentTask: null,
          history: [...current.history, action.result]
        });
        break;
    }
  }
}
```
