# 02. 아키텍처 패턴 분석

## 문서 목적

이 문서는 Claude Code의 핵심 아키텍처 패턴을 심층 분석합니다. 특히 의존성 주입(DI) 시스템, 이벤트 기반 통신, IPC 채널 메커니즘, 듀얼 프로세스 아키텍처의 구현 세부사항을 다룹니다. 이러한 패턴들은 hdsp-agent 개발 시 확장 가능하고 유지보수 가능한 시스템 설계의 기반이 됩니다.

## 1. 의존성 주입(DI) 시스템

### 1.1 서비스 식별자 패턴

Claude Code는 TypeScript 데코레이터와 브랜딩 패턴을 결합한 독특한 DI 시스템을 사용합니다.

#### 서비스 식별자 생성

```typescript
// /src/vs/platform/instantiation/common/instantiation.ts

/**
 * 서비스 식별자 생성 함수
 * - 런타임에 고유 ID를 생성하여 서비스 타입을 구분
 * - 컴파일 타임 타입 안전성을 제공
 */
export function createDecorator<T>(serviceId: string): ServiceIdentifier<T> {
  if (_util.serviceIds.has(serviceId)) {
    return _util.serviceIds.get(serviceId)!;
  }

  const id = <any>function (target: Function, key: string, index: number): any {
    if (arguments.length !== 3) {
      throw new Error('@IServiceName-decorator can only be used to decorate a parameter');
    }
    storeServiceDependency(id, target, index);
  };

  id.toString = () => serviceId;
  _util.serviceIds.set(serviceId, id);
  return id;
}

/**
 * 서비스 식별자 타입
 * - 함수 시그니처: 데코레이터로 사용 가능
 * - toString(): 디버깅용 문자열 표현
 */
export interface ServiceIdentifier<T> {
  (...args: any[]): void;
  type: T;
  toString(): string;
}
```

```typescript
// 2. 서비스 인터페이스 정의
export interface ICommandService {
  // 브랜드 마커: 타입 안전성 보장
  readonly _serviceBrand: undefined;

  // 서비스 메서드
  executeCommand<T = any>(commandId: string, ...args: any[]): Promise<T>;
  getCommands(): IEnumerator<string>;
  onWillExecuteCommand: Event<ICommandEvent>;
  onDidExecuteCommand: Event<ICommandEvent>;
}

/**
 * 브랜드 마커(_serviceBrand)의 역할:
 *
 * 1. 구조적 타입 시스템 보완
 *    - TypeScript는 구조적 타입을 사용 (duck typing)
 *    - 같은 구조의 두 인터페이스가 호환되는 문제 방지
 *
 * 2. 명시적 타입 구분
 *    - undefined 타입으로 런타임 오버헤드 없음
 *    - 컴파일 타임에만 타입 체킹에 사용
 *
 * 3. 실수 방지
 *    - 유사한 인터페이스를 잘못 사용하는 것을 방지
 */
```

#### 브랜딩 패턴의 중요성

```typescript
// 브랜딩이 없을 때의 문제
interface IServiceA {
  doSomething(): void;
}

interface IServiceB {
  doSomething(): void;
}

// TypeScript는 이 둘을 같은 타입으로 간주
function useA(service: IServiceA) {}
const serviceB: IServiceB = { doSomething: () => {} };
useA(serviceB); // ✅ 컴파일 통과 (의도하지 않은 동작)

// 브랜딩으로 해결
interface IServiceA {
  readonly _serviceBrand: undefined;
  doSomething(): void;
}

interface IServiceB {
  readonly _serviceBrand: undefined;
  doSomething(): void;
}

// 이제 타입 시스템이 구분 가능
useA(serviceB); // ❌ 컴파일 에러
```

```typescript
 *    - EditorService → [ICommandService, ILogService, IInstantiationService]
 *    ↓
 * 4. 인스턴스화 시 InstantiationService가 메타데이터 읽기
 *    ↓
 * 5. 각 서비스를 ServiceCollection에서 조회하여 주입
 */
```

#### 의존성 메타데이터 저장

```typescript
// /src/vs/platform/instantiation/common/instantiation.ts

const _util = {
  serviceIds: new Map<string, ServiceIdentifier<any>>(),
  DI_TARGET: '$di$target',
  DI_DEPENDENCIES: '$di$dependencies'
};

/**
 * 생성자 파라미터의 서비스 의존성 메타데이터 저장
 *
 * @param id - 서비스 식별자
 * @param target - 대상 클래스의 생성자
 * @param index - 파라미터 인덱스
 */
function storeServiceDependency(id: Function, target: Function, index: number): void {
  // 1. 기존 의존성 배열 가져오기 또는 생성
  if (target[_util.DI_TARGET] === target) {
    // 이미 메타데이터가 있음
    target[_util.DI_DEPENDENCIES].push({ id, index });
  } else {
    // 새로운 메타데이터 생성
    target[_util.DI_DEPENDENCIES] = [{ id, index }];
    target[_util.DI_TARGET] = target;
  }
}

/**
 * 저장된 메타데이터 구조:
 *
 * EditorService[_util.DI_DEPENDENCIES] = [
 *   { id: ICommandService, index: 0 },
 *   { id: ILogService, index: 1 },
 *   { id: IInstantiationService, index: 2 }
 * ]
 */
```

```typescript
// /src/vs/platform/instantiation/common/extensions.ts

/**
 * 인스턴스화 타이밍 제어
 */
export enum InstantiationType {
  /**
   * Eager: 즉시 인스턴스화
   * - 앱 시작 시 바로 생성
   * - 초기화 작업이 필요한 서비스
   * - 예: LogService, EnvironmentService
   */
  Eager = 0,

  /**
   * Delayed: 첫 요청 시 인스턴스화
   * - 메모리 최적화
   * - 대부분의 서비스가 사용
   * - 예: CommandService, ConfigurationService
   */
  Delayed = 1
}

/**
 * 서비스 등록 (싱글톤 패턴)
 *
 * @param id - 서비스 식별자
 * @param ctor - 구현 클래스
 * @param instantiationType - 인스턴스화 타이밍
 */
export function registerSingleton<T, Services extends BrandedService[]>(
  id: ServiceIdentifier<T>,
  ctor: { new(...services: Services): T },
  instantiationType?: InstantiationType
): void {
  // 서비스 디스크립터 생성
  const descriptor = new SyncDescriptor<T>(ctor, [], true);

  // 글로벌 서비스 레지스트리에 등록
  getSingletonServiceDescriptors().set(id, descriptor);

  // Eager 서비스는 즉시 인스턴스화 대기열에 추가
  if (instantiationType === InstantiationType.Eager) {
    getEagerServices().add(id);
  }
}

/**
 * 사용 예제:
 */
registerSingleton(ILogService, LogService, InstantiationType.Eager);
registerSingleton(ICommandService, CommandService, InstantiationType.Delayed);
```

### 1.4 InstantiationService 구현

```typescript
// /src/vs/platform/instantiation/common/instantiationService.ts

/**
 * 서비스 인스턴스화 엔진
 */
export class InstantiationService implements IInstantiationService {
  readonly _serviceBrand: undefined;

  private readonly _services: ServiceCollection;
  private readonly _strict: boolean;
  private readonly _parent?: InstantiationService;

  constructor(
    services: ServiceCollection = new ServiceCollection(),
    strict: boolean = false,
    parent?: InstantiationService
  ) {
    this._services = services;
    this._strict = strict;
    this._parent = parent;

    // 자기 자신을 서비스로 등록
    this._services.set(IInstantiationService, this);
  }

  /**
   * 인스턴스 생성 (메인 진입점)
   */
  createInstance<T>(ctorOrDescriptor: any, ...rest: any[]): T {
    let result: T;

    if (ctorOrDescriptor instanceof SyncDescriptor) {
      // 디스크립터에서 생성
      result = this._createInstance(ctorOrDescriptor.ctor, ctorOrDescriptor.staticArguments.concat(rest));
    } else {
      // 생성자에서 직접 생성
      result = this._createInstance(ctorOrDescriptor, rest);
    }

    return result;
  }

  /**
   * 실제 인스턴스 생성 로직
   */
  private _createInstance<T>(ctor: any, args: any[] = []): T {
    // 1. 의존성 메타데이터 가져오기
    const serviceDependencies = _util.getServiceDependencies(ctor).sort((a, b) => a.index - b.index);

    // 2. 서비스 인자 배열 생성
    const serviceArgs: any[] = [];
    for (const dependency of serviceDependencies) {
      const service = this._getOrCreateServiceInstance(dependency.id);
      if (!service && this._strict) {
        throw new Error(`[createInstance] ${ctor.name} depends on UNKNOWN service ${dependency.id}.`);
      }
      serviceArgs.push(service);
    }

    // 3. 정적 인자 + 서비스 인자 병합
    const firstServiceArgPos = serviceDependencies.length > 0 ? serviceDependencies[0].index : args.length;

    // 정적 인자는 앞에, 서비스 인자는 의존성 위치에 삽입
    if (args.length !== firstServiceArgPos) {
      console.trace(`[createInstance] First service dependency of ${ctor.name} at position ${firstServiceArgPos + 1} conflicts with ${args.length} static arguments`);
    }

    const finalArgs = args.slice(0, firstServiceArgPos).concat(serviceArgs);

    // 4. 인스턴스 생성
    return <T>new ctor(...finalArgs);
  }

  /**
   * 서비스 인스턴스 가져오기 또는 생성
   */
  private _getOrCreateServiceInstance<T>(id: ServiceIdentifier<T>): T {
    // 1. 현재 컬렉션에서 찾기
    let thing = this._services.get(id);

    // 2. 부모 컬렉션에서 찾기 (계층적 DI)
    if (!thing && this._parent) {
      return this._parent._getOrCreateServiceInstance(id);
    }

    // 3. 디스크립터인 경우 인스턴스화
    if (thing instanceof SyncDescriptor) {
      return this._createAndCacheServiceInstance(id, thing);
    } else {
      return thing;
    }
  }

  /**
   * 서비스 인스턴스 생성 및 캐싱
   */
  private _createAndCacheServiceInstance<T>(id: ServiceIdentifier<T>, desc: SyncDescriptor<T>): T {
    // 순환 의존성 감지
    const graph = new Graph<{ id: ServiceIdentifier<any>; desc: SyncDescriptor<any> }>((data) => data.id.toString());

    let cycleCount = 0;
    const stack = [{ id, desc }];

    while (stack.length) {
      const item = stack.pop()!;
      graph.lookupOrInsertNode(item);

      // 의존하는 서비스들을 그래프에 추가
      if (cycleCount++ > 1000) {
        throw new Error('Possible cycle in service dependencies');
      }

      const dependencies = _util.getServiceDependencies(item.desc.ctor);
      for (const dependency of dependencies) {
        const instanceOrDesc = this._services.get(dependency.id);
        if (instanceOrDesc instanceof SyncDescriptor) {
          // 그래프에 엣지 추가
          graph.insertEdge(item, { id: dependency.id, desc: instanceOrDesc });
          stack.push({ id: dependency.id, desc: instanceOrDesc });
        }
      }
    }

    // 순환 의존성 검사
    while (true) {
      const roots = graph.roots();

      if (roots.length === 0) {
        if (!graph.isEmpty()) {
          throw new Error('Cycle detected in service dependencies');
        }
        break;
      }

      for (const root of roots) {
        // 루트 노드부터 인스턴스 생성
        const instance = this._createInstance(root.data.desc.ctor, root.data.desc.staticArguments);
        this._services.set(root.data.id, instance);
        graph.removeNode(root.data);
      }
    }

    return this._services.get(id)!;
  }

  /**
   * 자식 컨테이너 생성 (계층적 DI)
   */
  createChild(services: ServiceCollection): IInstantiationService {
    return new InstantiationService(services, this._strict, this);
  }
}

/**
 * 의존성 메타데이터 유틸리티
 */
namespace _util {
  export function getServiceDependencies(ctor: any): { id: ServiceIdentifier<any>; index: number }[] {
    return ctor[DI_DEPENDENCIES] || [];
  }
}
```

#### 순환 의존성 감지 알고리즘

```typescript
/**
 * 그래프 기반 순환 의존성 감지
 *
 * 시나리오:
 * ServiceA → ServiceB → ServiceC
 *         ↖︎______________|
 *
 * 감지 과정:
 * 1. 각 서비스를 그래프 노드로 추가
 * 2. 의존성을 엣지로 연결
 * 3. 루트 노드(진입 차수 0) 탐색
 * 4. 루트가 없고 그래프가 비어있지 않으면 → 순환
 */

class Graph<T> {
  private nodes = new Map<string, Node<T>>();

  constructor(private readonly _hashFn: (element: T) => string) {}

  lookupOrInsertNode(data: T): Node<T> {
    const key = this._hashFn(data);
    let node = this.nodes.get(key);

    if (!node) {
      node = new Node(data);
      this.nodes.set(key, node);
    }

    return node;
  }

  insertEdge(from: T, to: T): void {
    const fromNode = this.lookupOrInsertNode(from);
    const toNode = this.lookupOrInsertNode(to);

    fromNode.outgoing.add(toNode);
    toNode.incoming.add(fromNode);
  }

  removeNode(data: T): void {
    const key = this._hashFn(data);
    const node = this.nodes.get(key);

    if (!node) return;

    // 진입/진출 엣지 제거
    for (const outgoing of node.outgoing) {
      outgoing.incoming.delete(node);
    }
    for (const incoming of node.incoming) {
      incoming.outgoing.delete(node);
    }

    this.nodes.delete(key);
  }

  roots(): Node<T>[] {
    const result: Node<T>[] = [];
    for (const node of this.nodes.values()) {
      if (node.incoming.size === 0) {
        result.push(node);
      }
    }
    return result;
  }

  isEmpty(): boolean {
    return this.nodes.size === 0;
  }
}

class Node<T> {
  readonly incoming = new Set<Node<T>>();
  readonly outgoing = new Set<Node<T>>();

  constructor(readonly data: T) {}
}
```

### 1.5 ServiceCollection 구현

```typescript
// /src/vs/platform/instantiation/common/serviceCollection.ts

/**
 * 서비스 레지스트리
 * - Map 기반 서비스 저장소
 * - 인스턴스 또는 디스크립터 저장
 */
export class ServiceCollection implements IServiceCollection {
  private _entries = new Map<ServiceIdentifier<any>, any>();

  constructor(...entries: [ServiceIdentifier<any>, any][]) {
    for (const [id, service] of entries) {
      this.set(id, service);
    }
  }

  set<T>(id: ServiceIdentifier<T>, instanceOrDescriptor: T | SyncDescriptor<T>): T | SyncDescriptor<T> {
    const result = this._entries.get(id);
    this._entries.set(id, instanceOrDescriptor);
    return result;
  }

  has(id: ServiceIdentifier<any>): boolean {
    return this._entries.has(id);
  }

  get<T>(id: ServiceIdentifier<T>): T | SyncDescriptor<T> {
    return this._entries.get(id);
  }
}

 * 사용 예제:
 */
const services = new ServiceCollection(
  [ILogService, new LogService()],
  [ICommandService, new SyncDescriptor(CommandService)]
);

const instantiationService = new InstantiationService(services);
const myFeature = instantiationService.createInstance(MyFeature);
```

### 2.1 Event 인터페이스 설계

```typescript
// /src/vs/base/common/event.ts

/**
 * 이벤트 인터페이스: 호출 가능한 함수 타입
 *
 * 핵심 아이디어:
 * - 이벤트는 "구독 함수"
 * - 호출하면 구독이 시작되고 Disposable 반환
 * - 메모리 누수 방지를 위한 명시적 구독 해제
 */
export interface Event<T> {
  /**
   * @param listener - 이벤트 핸들러
   * @param thisArgs - 핸들러 내부의 this 컨텍스트
   * @param disposables - 구독을 저장할 배열 (자동 해제용)
   * @returns 구독 해제용 Disposable
   */
  (listener: (e: T) => any, thisArgs?: any, disposables?: IDisposable[]): IDisposable;
}

/**
 * 사용 예제:
 */
interface FileChangeEvent {
  path: string;
  type: 'create' | 'update' | 'delete';
}

declare const onDidChangeFile: Event<FileChangeEvent>;

// 1. 기본 구독
const disposable = onDidChangeFile((e) => {
  console.log(`File ${e.type}: ${e.path}`);
});

// 나중에 구독 해제
disposable.dispose();

// 2. this 컨텍스트 지정
class FileWatcher {
  private handleFileChange(e: FileChangeEvent) {
    // this는 FileWatcher 인스턴스
```

### 2.2 Emitter 구현

```typescript
// /src/vs/base/common/event.ts

/**
 * 이벤트 발행자
 * - 이벤트를 발행(fire)하고 리스너를 관리
 */
export class Emitter<T> {
  private _event?: Event<T>;
  private _listeners?: LinkedList<Listener<T>>;
  private _deliveryQueue?: EventDeliveryQueue;
  protected _disposed: boolean = false;

  /**
   * 이벤트 접근자
   * - 지연 초기화: 첫 구독 시점에 생성
   */
  get event(): Event<T> {
    if (!this._event) {
      this._event = (listener: (e: T) => any, thisArgs?: any, disposables?: IDisposable[]) => {
        // 리스너 목록 초기화
        if (!this._listeners) {
          this._listeners = new LinkedList();
        }

        // 리스너 등록
        const remove = this._listeners.push(!thisArgs ? listener : [listener, thisArgs]);

        // Disposable 생성
        const result: IDisposable = {
          dispose: () => {
            if (!this._disposed) {
              remove();
            }
          }
        };

        // 자동 해제 배열에 추가
        if (disposables) {
          disposables.push(result);
        }

        return result;
      };
    }

    return this._event;
  }

  /**
   * 이벤트 발행
   * - 모든 리스너에게 이벤트 전달
   */
  fire(event: T): void {
    if (this._listeners) {
      // 전달 큐가 없으면 즉시 실행
      if (!this._deliveryQueue) {
        this._deliveryQueue = new EventDeliveryQueue();
      }

      for (const listener of this._listeners) {
        this._deliveryQueue.push(this, listener, event);
      }

      this._deliveryQueue.deliver();
    }
  }

  /**
   * 리소스 해제
   */
  dispose(): void {
    if (!this._disposed) {
      this._disposed = true;

      // 리스너 목록 클리어
      if (this._listeners) {
        this._listeners.clear();
        this._listeners = undefined;
      }

      // 전달 큐 클리어
      if (this._deliveryQueue) {
        this._deliveryQueue.clear(this);
      }
    }
  }
}

/**
 * 리스너 타입
 */
type Listener<T> = [(e: T) => void, any] | ((e: T) => void);

/**
 * 이벤트 전달 큐 (마이크로태스크 최적화)
 */
class EventDeliveryQueue {
  private _queue = new LinkedList<EventDeliveryQueueElement>();

  push<T>(emitter: Emitter<T>, listener: Listener<T>, event: T): void {
    this._queue.push({ emitter, listener, event });
  }

  deliver(): void {
    while (this._queue.size > 0) {
      const element = this._queue.shift()!;
      try {
        if (Array.isArray(element.listener)) {
          const [callback, thisArgs] = element.listener;
          callback.call(thisArgs, element.event);
        } else {
          element.listener(element.event);
        }
      } catch (e) {
        console.error(e);
      }
    }
  }

  clear<T>(emitter: Emitter<T>): void {
    // 특정 emitter의 이벤트만 제거
    const newQueue = new LinkedList<EventDeliveryQueueElement>();
    for (const element of this._queue) {
      if (element.emitter !== emitter) {
        newQueue.push(element);
      }
    }
    this._queue = newQueue;
  }
}

interface EventDeliveryQueueElement {
  emitter: Emitter<any>;
  listener: Listener<any>;
  event: any;
}
```

```typescript
// /src/vs/base/common/event.ts

/**
 * Event 네임스페이스: 유틸리티 함수 모음
 */
export namespace Event {
  /**
   * 없는 이벤트 (항상 발생하지 않음)
   */
  export const None: Event<any> = () => Disposable.None;

  /**
   * 한 번만 발생하는 이벤트
   */
  export function once<T>(event: Event<T>): Event<T> {
    return (listener, thisArgs = null, disposables?) => {
      let didFire = false;
      const result = event((e) => {
        if (didFire) {
          return;
        }
        didFire = true;

        result.dispose(); // 자동 구독 해제

        return listener.call(thisArgs, e);
      }, null, disposables);

      return result;
    };
  }

  /**
   * 이벤트 값 변환
   */
  export function map<I, O>(event: Event<I>, map: (i: I) => O): Event<O> {
    return (listener, thisArgs = null, disposables?) => {
      return event((i) => listener.call(thisArgs, map(i)), null, disposables);
    };
  }

  /**
   * 이벤트 필터링
   */
  export function filter<T>(event: Event<T>, filter: (e: T) => boolean): Event<T>;
  export function filter<T, U>(event: Event<T | U>, filter: (e: T | U) => e is T): Event<T>;
  export function filter<T>(event: Event<T>, filter: (e: T) => boolean): Event<T> {
    return (listener, thisArgs = null, disposables?) => {
      return event((e) => filter(e) && listener.call(thisArgs, e), null, disposables);
    };
  }

  /**
   * 디바운스 (마지막 이벤트만 전달)
   */
  export function debounce<T>(
    event: Event<T>,
    merge: (last: T | undefined, event: T) => T,
    delay: number = 100,
    leading: boolean = false,
    leakWarningThreshold?: number
  ): Event<T> {
    let subscription: IDisposable;
    let output: Emitter<T> | undefined = undefined;
    let handle: any = undefined;
    let numDebouncedCalls = 0;

    const emitter = new Emitter<T>({
      onFirstListenerAdd() {
        subscription = event((cur) => {
          numDebouncedCalls++;
          const merged = merge(output?.event as any, cur);

          if (leading && !handle) {
            emitter.fire(merged);
          } else {
            clearTimeout(handle);
            handle = setTimeout(() => {
              const merged = merge(output?.event as any, cur);
              emitter.fire(merged);
              numDebouncedCalls = 0;
            }, delay);
          }
        });
      },
      onLastListenerRemove() {
        subscription.dispose();
      }
    });

    return emitter.event;
  }

  /**
   * 여러 이벤트를 하나로 병합
   */
  export function any<T>(...events: Event<T>[]): Event<T> {
    return (listener, thisArgs = null, disposables?) => {
      const disposable = combinedDisposable(...events.map((event) => event((e) => listener.call(thisArgs, e))));

      if (disposables) {
        disposables.push(disposable);
      }

      return disposable;
    };
  }

  /**
   * 조건부 이벤트 발행
   */
  export function signal<T>(event: Event<T>): Event<void> {
    return event as Event<any> as Event<void>;
  }

  /**
   * 버퍼링 (여러 이벤트를 모아서 한 번에 전달)
   */
  export function buffer<T>(event: Event<T>, flushAfterTimeout = false, _buffer: T[] = []): Event<T> {
    let buffer: T[] | null = _buffer.slice();

    const emitter = new Emitter<T>({
      onFirstListenerAdd() {
        if (!buffer) {
          buffer = _buffer.slice();
        }

        const listener = event((e) => {
          if (buffer) {
            buffer.push(e);
          } else {
            emitter.fire(e);
          }
        });
      },
      onFirstListenerDidAdd() {
        if (buffer) {
          buffer.forEach((e) => emitter.fire(e));
          buffer = null;
        }
      }
    });

    return emitter.event;
  }
}

 * 사용 예제:
 */
```

### 2.4 EventMultiplexer 패턴

```typescript
/**
 * 동적 이벤트 소스 관리
 * - 여러 이벤트 소스를 하나의 이벤트로 통합
 * - 소스 추가/제거 시 자동 구독 관리
 */
export class EventMultiplexer<T> implements IDisposable {
  private readonly emitter: Emitter<T>;
  readonly event: Event<T>;

  private readonly hasListeners = false;
  private readonly events: { event: Event<T>; disposable: IDisposable }[] = [];

  constructor() {
    this.emitter = new Emitter<T>({
      onFirstListenerAdd: () => this.onFirstListenerAdd(),
      onLastListenerRemove: () => this.onLastListenerRemove()
    });

    this.event = this.emitter.event;
  }

  add(event: Event<T>): IDisposable {
    const e = { event, disposable: Disposable.None };
    this.events.push(e);

    if (this.hasListeners) {
      this.hook(e);
    }

    const dispose = () => {
      if (this.hasListeners) {
        this.unhook(e);
      }

      const idx = this.events.indexOf(e);
      this.events.splice(idx, 1);
    };

    return toDisposable(dispose);
  }

  private onFirstListenerAdd(): void {
    this.hasListeners = true;
    this.events.forEach((e) => this.hook(e));
  }

  private onLastListenerRemove(): void {
    this.hasListeners = false;
    this.events.forEach((e) => this.unhook(e));
  }

  private hook(e: { event: Event<T>; disposable: IDisposable }): void {
    e.disposable = e.event((r) => this.emitter.fire(r));
  }

  private unhook(e: { event: Event<T>; disposable: IDisposable }): void {
    e.disposable.dispose();
    e.disposable = Disposable.None;
  }

  dispose(): void {
    this.emitter.dispose();
  }
}

/**
 * 사용 예제: 여러 파일 워처 통합
 */
class MultiFileWatcher {
  private readonly _onDidChange = new EventMultiplexer<FileChangeEvent>();
  readonly onDidChange: Event<FileChangeEvent> = this._onDidChange.event;

  private watchers = new Map<string, FileSystemWatcher>();

  watchFile(path: string): void {
    const watcher = new FileSystemWatcher();
    watcher.watchFile(path);

    // 워처의 이벤트를 멀티플렉서에 추가
    this._onDidChange.add(watcher.onDidChange);
    this.watchers.set(path, watcher);
  }

  unwatchFile(path: string): void {
    const watcher = this.watchers.get(path);
    if (watcher) {
      watcher.dispose();
      this.watchers.delete(path);
    }
  }

  dispose(): void {
    this._onDidChange.dispose();
    this.watchers.forEach((w) => w.dispose());
  }
}

});
```

### 3.3 클라이언트 측 채널

```typescript
// /src/vs/base/parts/ipc/common/ipc.ts

/**
 * 클라이언트 채널 인터페이스
 */
export interface IChannel {
  /**
   * 명령 호출 (Promise 기반)
   * @param command - 명령 이름
   * @param arg - 인자
   * @param cancellationToken - 취소 토큰
   */
  call<T>(command: string, arg?: any, cancellationToken?: CancellationToken): Promise<T>;

  /**
   * 이벤트 구독
   * @param event - 이벤트 이름
   * @param arg - 인자
   */
  listen<T>(event: string, arg?: any): Event<T>;
}

/**
 * 채널 클라이언트 구현
 */
export class ChannelClient implements IChannelClient, IDisposable {
  private lastRequestId = 0;
  private readonly protocolListener: IDisposable;
  private readonly activeRequests = new Map<number, IDisposable>();
  private readonly handlers = new Map<number, IHandler>();

  constructor(private readonly protocol: IMessagePassingProtocol) {
    this.protocolListener = this.protocol.onMessage((msg) => this.onMessage(msg));
  }

  getChannel<T extends IChannel>(channelName: string): T {
    const that = this;

    return {
      call(command: string, arg?: any, cancellationToken?: CancellationToken): Promise<any> {
        return that.requestPromise(channelName, command, arg, cancellationToken);
      },
      listen(event: string, arg?: any): Event<any> {
        return that.requestEvent(channelName, event, arg);
      }
    } as T;
  }

  /**
   * Promise 요청
   */
  private requestPromise(
    channelName: string,
    name: string,
    arg?: any,
    cancellationToken?: CancellationToken
  ): Promise<any> {
    const id = this.lastRequestId++;
    const request: IRawRequest = { id, type: RequestType.Promise, channelName, name, arg };

    // 취소 처리
    if (cancellationToken) {
      const disposable = cancellationToken.onCancellationRequested(() => {
        this.protocol.send({ id, type: RequestType.PromiseCancel });
      });
      this.activeRequests.set(id, disposable);
    }

    // Promise 생성
    const promise = new Promise((resolve, reject) => {
      this.handlers.set(id, { resolve, reject });
    });

    // 메시지 전송
    this.protocol.send(request);

    return promise.finally(() => {
      this.activeRequests.get(id)?.dispose();
      this.activeRequests.delete(id);
    });
  }

  /**
   * Event 요청
   */
  private requestEvent(channelName: string, name: string, arg?: any): Event<any> {
    const id = this.lastRequestId++;
    const emitter = new Emitter<any>();

    // 이벤트 구독 시작
    const request: IRawEventListen = { id, type: RequestType.EventListen, channelName, name, arg };
    this.protocol.send(request);

    // 핸들러 등록
    this.handlers.set(id, {
      resolve: (data: any) => emitter.fire(data),
      reject: (err: any) => emitter.fire(err)
    });

    // Disposable 반환
    return Event.once(Event.any(
      emitter.event,
      Event.fromCallback(() => {
        this.protocol.send({ id, type: RequestType.EventDispose });
        this.handlers.delete(id);
      })
    ))(emitter.event);
  }

  /**
   * 메시지 수신 처리
   */
  private onMessage(msg: IRawResponse | IRawEventFire): void {
    const handler = this.handlers.get(msg.id);
    if (!handler) return;

    if ('error' in msg && msg.error) {
      handler.reject(msg.error);
    } else {
      handler.resolve(msg.data);
    }
  }

  dispose(): void {
    this.protocolListener.dispose();
    this.handlers.clear();
    this.activeRequests.forEach((d) => d.dispose());
    this.activeRequests.clear();
  }
}

interface IHandler {
  resolve: (data: any) => void;
  reject: (err: any) => void;
}
```

```typescript
// /src/vs/base/parts/ipc/common/ipc.ts

/**
 * 서버 채널 인터페이스
 */
export interface IServerChannel<TContext = string> {
  /**
   * 명령 호출
   * @param ctx - 컨텍스트 (클라이언트 식별자)
   * @param command - 명령 이름
   * @param arg - 인자
   * @param cancellationToken - 취소 토큰
   */
  call<T>(ctx: TContext, command: string, arg?: any, cancellationToken?: CancellationToken): Promise<T>;

  /**
   * 이벤트 제공
   * @param ctx - 컨텍스트
   * @param event - 이벤트 이름
   * @param arg - 인자
   */
  listen<T>(ctx: TContext, event: string, arg?: any): Event<T>;
}

/**
 * 채널 서버 구현
 */
export class ChannelServer<TContext = string> implements IChannelServer<TContext>, IDisposable {
  private readonly channels = new Map<string, IServerChannel<TContext>>();
  private readonly protocolListener: IDisposable;
  private readonly activeRequests = new Map<number, IDisposable>();

  constructor(
    private readonly protocol: IMessagePassingProtocol,
    private readonly ctx: TContext
  ) {
    this.protocolListener = this.protocol.onMessage((msg) => this.onMessage(msg));
  }

  registerChannel(channelName: string, channel: IServerChannel<TContext>): void {
    this.channels.set(channelName, channel);
  }

  /**
   * 메시지 수신 처리
   */
  private onMessage(msg: IRawRequest): void {
    switch (msg.type) {
      case RequestType.Promise:
        this.onPromise(msg);
        break;
      case RequestType.PromiseCancel:
        this.onPromiseCancel(msg);
        break;
      case RequestType.EventListen:
        this.onEventListen(msg);
        break;
      case RequestType.EventDispose:
        this.onEventDispose(msg);
        break;
    }
  }

  /**
   * Promise 요청 처리
   */
  private onPromise(msg: IRawRequest): void {
    const channel = this.channels.get(msg.channelName);
    if (!channel) {
      this.sendResponse(msg.id, { error: { message: `Channel not found: ${msg.channelName}` } });
      return;
    }

    const cancellationTokenSource = new CancellationTokenSource();
    this.activeRequests.set(msg.id, cancellationTokenSource);

    // 채널 메서드 호출
    channel
      .call(this.ctx, msg.name, msg.arg, cancellationTokenSource.token)
      .then(
        (data) => {
          this.sendResponse(msg.id, { data });
          this.activeRequests.delete(msg.id);
        },
        (err) => {
          this.sendResponse(msg.id, {
            error: {
              message: err.message,
              name: err.name,
              stack: err.stack ? err.stack.split('\n') : []
            }
          });
          this.activeRequests.delete(msg.id);
        }
      );
  }

  /**
   * Promise 취소 처리
   */
  private onPromiseCancel(msg: IRawRequest): void {
    const cancellationTokenSource = this.activeRequests.get(msg.id);
    if (cancellationTokenSource) {
      cancellationTokenSource.cancel();
      this.activeRequests.delete(msg.id);
    }
  }

  /**
   * Event 구독 처리
   */
  private onEventListen(msg: IRawEventListen): void {
    const channel = this.channels.get(msg.channelName);
    if (!channel) {
      this.sendResponse(msg.id, { error: { message: `Channel not found: ${msg.channelName}` } });
      return;
    }

    // 이벤트 구독
    const event = channel.listen(this.ctx, msg.name, msg.arg);
    const disposable = event((data) => {
      this.sendResponse(msg.id, { data });
    });

    this.activeRequests.set(msg.id, disposable);
  }

  /**
   * Event 구독 해제 처리
   */
  private onEventDispose(msg: IRawRequest): void {
    const disposable = this.activeRequests.get(msg.id);
    if (disposable) {
      disposable.dispose();
      this.activeRequests.delete(msg.id);
    }
  }

  /**
   * 응답 전송
   */
  private sendResponse(id: number, response: { data?: any; error?: any }): void {
    this.protocol.send({ id, ...response });
  }

  dispose(): void {
    this.protocolListener.dispose();
    this.activeRequests.forEach((d) => d.dispose());
    this.activeRequests.clear();
  }
}
```

### 4.2 DisposableStore

```typescript
/**
 * Disposable 컬렉션 관리
 */
export class DisposableStore implements IDisposable {
  static DISABLE_DISPOSED_WARNING = false;

  private _toDispose = new Set<IDisposable>();
  private _isDisposed = false;

  /**
   * Disposable 추가
   */
  public add<T extends IDisposable>(t: T): T {
    if (!t) {
      return t;
    }
    if ((t as unknown as DisposableStore) === this) {
      throw new Error('Cannot register a disposable on itself!');
    }

    if (this._isDisposed) {
      if (!DisposableStore.DISABLE_DISPOSED_WARNING) {
        console.warn('Trying to add a disposable to a DisposableStore that has already been disposed');
      }
      t.dispose();
    } else {
      this._toDispose.add(t);
    }

    return t;
  }

  /**
   * Disposable 제거 (해제하지 않음)
   */
  public delete<T extends IDisposable>(t: T): void {
    this._toDispose.delete(t);
  }

  /**
   * 전체 해제
   */
  public dispose(): void {
    if (this._isDisposed) {
      return;
    }

    markAsDisposed(this);
    this._isDisposed = true;
    this.clear();
  }

  /**
   * 전체 제거 및 해제
   */
  public clear(): void {
    if (this._toDispose.size === 0) {
      return;
    }

    try {
      dispose(this._toDispose);
    } finally {
      this._toDispose.clear();
    }
  }
}

/**
 * 배열/이터러블 일괄 해제
 */
export function dispose<T extends IDisposable>(disposables: Iterable<T>): T[];
export function dispose<T extends IDisposable>(disposables: T[]): T[];
export function dispose<T extends IDisposable>(arg: T | Iterable<T> | undefined): any {
  if (!arg) return [];

  if (Symbol.iterator in arg) {
    const errors: any[] = [];

    for (const d of arg) {
      if (d) {
        try {
          d.dispose();
        } catch (e) {
          errors.push(e);
        }
      }
    }

    if (errors.length === 1) {
      throw errors[0];
    } else if (errors.length > 1) {
      throw new AggregateError(errors, 'Encountered errors while disposing of store');
    }

    return Array.isArray(arg) ? [] : arg;
  } else {
    arg.dispose();
    return arg;
  }
}
```

### 4.3 MutableDisposable 패턴

```typescript
/**
 * 교체 가능한 Disposable
 * - 이전 값을 자동으로 해제하고 새 값으로 교체
 */
export class MutableDisposable<T extends IDisposable> implements IDisposable {
  private _value?: T;
  private _isDisposed = false;

  constructor() {}

  get value(): T | undefined {
    return this._isDisposed ? undefined : this._value;
  }

  set value(value: T | undefined) {
    if (this._isDisposed) {
      value?.dispose();
    } else {
      this._value?.dispose();
      this._value = value;
    }
  }

  /**
   * 값 교체 (명시적 메서드)
   */
  clear(): void {
    this.value = undefined;
  }

  dispose(): void {
    this._isDisposed = true;
    markAsDisposed(this);
    this._value?.dispose();
    this._value = undefined;
  }
}

/**
 * 사용 예제:
 */
class ComponentWithSubscription extends Disposable {
  private readonly fileWatcher = this._register(new MutableDisposable<IDisposable>());

  watchFile(path: string): void {

      if (e.path === path) {
        console.log('File changed');
      }
    });
  }

  stopWatching(): void {
    this.fileWatcher.clear(); // 구독 해제
  }
}
```

```typescript
/**

export function combinedDisposable(...disposables: IDisposable[]): IDisposable {
  const parent = toDisposable(() => dispose(disposables));
  setParentOfDisposables(disposables, parent);
  return parent;
}

/**
 * 함수를 Disposable로 변환
 */
export function toDisposable(fn: () => void): IDisposable {
  const self = {
    dispose: () => {
      markAsDisposed(self);
      fn();
    }
  };
  return self;
}
```

```typescript
class AgentServerChannel implements IServerChannel {
  constructor(
    @IAgentService private readonly agentService: IAgentService,
    @IToolService private readonly toolService: IToolService
  ) {}

  async call(ctx: string, command: string, arg?: any): Promise<any> {
    switch (command) {
      case 'executeTask':
        return this.agentService.executeTask(arg);
      case 'delegateTask':
        return this.agentService.delegateToAgent(arg.agentId, arg.task);
      case 'executeTool':
        return this.toolService.executeTool(arg.toolId, arg.args);
      default:
        throw new Error(`Unknown command: ${command}`);
    }
  }

  listen(ctx: string, event: string, arg?: any): Event<any> {

    }
  }
```
