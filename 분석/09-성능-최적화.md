## 2. LRU 캐싱

### 2.1 LRU 캐시 구현

```typescript
class LRUCache<K, V> {
  private readonly cache = new Map<K, V>();
  private readonly maxSize: number;

  constructor(maxSize: number) {
    this.maxSize = maxSize;
  }

  get(key: K): V | undefined {
    const value = this.cache.get(key);
    if (value !== undefined) {
      // 최근 사용으로 이동
      this.cache.delete(key);
      this.cache.set(key, value);
    }
    return value;
  }

  set(key: K, value: V): void {
    // 이미 존재하면 삭제 (순서 갱신)
    if (this.cache.has(key)) {
      this.cache.delete(key);
    }
    // 용량 초과 시 가장 오래된 항목 삭제
    else if (this.cache.size >= this.maxSize) {
      const oldest = this.cache.keys().next().value;
      this.cache.delete(oldest);
    }

    this.cache.set(key, value);
  }

  has(key: K): boolean {
    return this.cache.has(key);
  }

  delete(key: K): boolean {
    return this.cache.delete(key);
  }

  clear(): void {
    this.cache.clear();
  }

  get size(): number {
    return this.cache.size;
  }
}
```

### 2.2 TTL 지원 캐시

```typescript
interface CacheEntry<V> {
  value: V;
  expiry: number;
}

class TTLCache<K, V> {
  private readonly cache = new Map<K, CacheEntry<V>>();
  private readonly defaultTTL: number;

  constructor(defaultTTL: number = 60000) {
    this.defaultTTL = defaultTTL;
    this.startCleanup();
  }

  get(key: K): V | undefined {
    const entry = this.cache.get(key);
    if (!entry) return undefined;

    if (Date.now() > entry.expiry) {
      this.cache.delete(key);
      return undefined;
    }

    return entry.value;
  }

  set(key: K, value: V, ttl?: number): void {
    this.cache.set(key, {
      value,
      expiry: Date.now() + (ttl ?? this.defaultTTL)
    });
  }

  private startCleanup(): void {
    setInterval(() => {
      const now = Date.now();
      for (const [key, entry] of this.cache) {
        if (now > entry.expiry) {
          this.cache.delete(key);
        }
      }
    }, 60000);
  }
}
```

## 3. 스로틀링 / 디바운싱

### 3.1 ThrottledDelayer

```typescript
class ThrottledDelayer<T> {
  private timeout: NodeJS.Timeout | undefined;
  private completionPromise: Promise<T> | undefined;
  private resolve: ((value: T) => void) | undefined;
  private fn: (() => Promise<T>) | undefined;

  constructor(private defaultDelay: number) {}

  trigger(fn: () => Promise<T>, delay?: number): Promise<T> {
    this.fn = fn;

    if (!this.completionPromise) {
      this.completionPromise = new Promise<T>(resolve => {
        this.resolve = resolve;
      });
    }

    this.cancelTimeout();
    this.timeout = setTimeout(() => {
      this.timeout = undefined;
      this.doExecute();
    }, delay ?? this.defaultDelay);

    return this.completionPromise;
  }

  private async doExecute(): Promise<void> {
    if (this.fn) {
      const result = await this.fn();
      this.resolve?.(result);
      this.completionPromise = undefined;
      this.resolve = undefined;
    }
  }

  private cancelTimeout(): void {
    if (this.timeout) {
      clearTimeout(this.timeout);
      this.timeout = undefined;
    }
  }

  cancel(): void {
    this.cancelTimeout();
    this.completionPromise = undefined;
    this.resolve = undefined;
    this.fn = undefined;
  }

  dispose(): void {
    this.cancel();
  }
}
```

### 3.2 Debounce 함수

```typescript
function debounce<T extends (...args: any[]) => any>(
  fn: T,
  delay: number
): T & { cancel: () => void } {
  let timeout: NodeJS.Timeout | undefined;

  const debounced = function(this: any, ...args: Parameters<T>) {
    if (timeout) {
      clearTimeout(timeout);
    }

    timeout = setTimeout(() => {
      fn.apply(this, args);
      timeout = undefined;
    }, delay);
  } as T & { cancel: () => void };

  debounced.cancel = () => {
    if (timeout) {
      clearTimeout(timeout);
      timeout = undefined;
    }
  };

  return debounced;
}

// 사용 예시
const debouncedSearch = debounce((query: string) => {
  performSearch(query);
}, 300);
```

### 3.3 Throttle 함수

```typescript
function throttle<T extends (...args: any[]) => any>(
  fn: T,
  limit: number
): T {
  let lastCall = 0;
  let timeout: NodeJS.Timeout | undefined;
  let lastArgs: Parameters<T> | undefined;

  return function(this: any, ...args: Parameters<T>) {
    const now = Date.now();
    const remaining = limit - (now - lastCall);

    lastArgs = args;

    if (remaining <= 0) {
      if (timeout) {
        clearTimeout(timeout);
        timeout = undefined;
      }
      lastCall = now;
      fn.apply(this, args);
    } else if (!timeout) {
      timeout = setTimeout(() => {
        lastCall = Date.now();
        timeout = undefined;
        if (lastArgs) {
          fn.apply(this, lastArgs);
        }
      }, remaining);
    }
  } as T;
}
```

## 4. Sequencer 패턴

### 4.1 순차 실행 보장

```typescript
class Sequencer {
  private queue: Promise<any> = Promise.resolve();

  queue<T>(promiseTask: () => Promise<T>): Promise<T> {
    return this.queue = this.queue.then(
      () => promiseTask(),
      () => promiseTask()
    );
  }
}

// 사용 예시
const sequencer = new Sequencer();

// 동시에 호출되어도 순차 실행
sequencer.queue(() => saveFile('a.txt'));
sequencer.queue(() => saveFile('b.txt'));
sequencer.queue(() => saveFile('c.txt'));
```

### 4.2 MCP 서버 연결 관리

```typescript
class McpServer {
  private readonly _connectionSequencer = new Sequencer();

  public start(): Promise<McpConnectionState> {
    return this._connectionSequencer.queue(async () => {
      if (this.isRunning) {
        return McpConnectionState.Running;
      }

      this.setState(McpConnectionState.Starting);

      try {
        await this.connect();
        this.setState(McpConnectionState.Running);
        return McpConnectionState.Running;
      } catch (error) {
        this.setState(McpConnectionState.Error);
        throw error;
      }
    });
  }

  public stop(): Promise<void> {
    return this._connectionSequencer.queue(async () => {
      if (!this.isRunning) {
        return;
      }

      await this.disconnect();
      this.setState(McpConnectionState.Stopped);
    });
  }
}
```

### 5.2 병렬 실행 제한

```typescript
async function parallelWithLimit<T, R>(
  items: T[],
  fn: (item: T) => Promise<R>,
  limit: number
): Promise<R[]> {
  const results: R[] = [];
  const executing: Promise<void>[] = [];

  for (const item of items) {
    const p = fn(item).then(result => {
      results.push(result);
    });

    executing.push(p);

    if (executing.length >= limit) {
      await Promise.race(executing);
      // 완료된 것 제거
      const completed = executing.findIndex(
        e => e === await Promise.race([e, Promise.resolve(null)])
      );
      if (completed >= 0) {
        executing.splice(completed, 1);
      }
    }
  }

  await Promise.all(executing);
  return results;
}

// 사용 예시
const results = await parallelWithLimit(
  files,
  file => processFile(file),
  5 // 최대 5개 동시 실행
);
```

### 5.3 배치 처리

```typescript
function batch<T>(items: T[], size: number): T[][] {
  const batches: T[][] = [];
  for (let i = 0; i < items.length; i += size) {
    batches.push(items.slice(i, i + size));
  }
  return batches;
}

async function processBatches<T, R>(
  items: T[],
  fn: (batch: T[]) => Promise<R[]>,
  batchSize: number
): Promise<R[]> {
  const batches = batch(items, batchSize);
  const results: R[] = [];

  for (const batchItems of batches) {
    const batchResults = await fn(batchItems);
    results.push(...batchResults);
  }

  return results;
}
```

## 6. 지연 로딩

### 6.1 Lazy 클래스

```typescript
class Lazy<T> {
  private _value?: T;
  private _isValueCreated = false;

  constructor(private readonly factory: () => T) {}

  get value(): T {
    if (!this._isValueCreated) {
      this._value = this.factory();
      this._isValueCreated = true;
    }
    return this._value!;
  }

  get isValueCreated(): boolean {
    return this._isValueCreated;
  }
}

// 사용 예시
const expensiveData = new Lazy(() => computeExpensiveData());

// 필요할 때만 계산
if (condition) {
  const data = expensiveData.value;
}
```

### 6.3 서비스 지연 초기화

```typescript
class LazyService<T> {
  private _instance?: T;
  private _initializing?: Promise<T>;

  constructor(
    private readonly factory: () => Promise<T>
  ) {}

  async get(): Promise<T> {
    if (this._instance) {
      return this._instance;
    }

    if (this._initializing) {
      return this._initializing;
    }

    this._initializing = this.factory().then(instance => {
      this._instance = instance;
      this._initializing = undefined;
      return instance;
    });

    return this._initializing;
  }
}
```

## 7. 메모이제이션

### 7.1 함수 메모이제이션

```typescript
function memoize<T extends (...args: any[]) => any>(
  fn: T,
  keyFn?: (...args: Parameters<T>) => string
): T {
  const cache = new Map<string, ReturnType<T>>();

  return function(this: any, ...args: Parameters<T>): ReturnType<T> {
    const key = keyFn ? keyFn(...args) : JSON.stringify(args);

    if (cache.has(key)) {
      return cache.get(key)!;
    }

    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  } as T;
}

// 사용 예시
const expensiveCalculation = memoize((a: number, b: number) => {
  // 복잡한 계산
  return a * b;
});
```

### 7.2 비동기 메모이제이션

```typescript
function memoizeAsync<T extends (...args: any[]) => Promise<any>>(
  fn: T,
  options?: {
    maxSize?: number;
    ttl?: number;
    keyFn?: (...args: Parameters<T>) => string;
  }
): T {
  const cache = new LRUCache<string, { value: any; expiry: number }>(
    options?.maxSize ?? 100
  );
  const ttl = options?.ttl ?? Infinity;

  return async function(this: any, ...args: Parameters<T>) {
    const key = options?.keyFn
      ? options.keyFn(...args)
      : JSON.stringify(args);

    const cached = cache.get(key);
    if (cached && Date.now() < cached.expiry) {
      return cached.value;
    }

    const result = await fn.apply(this, args);
    cache.set(key, { value: result, expiry: Date.now() + ttl });
    return result;
  } as T;
}
```

## 8. 이벤트 최적화

### 8.1 이벤트 버퍼링

```typescript
class EventBuffer<T> {
  private buffer: T[] = [];
  private timeout?: NodeJS.Timeout;

  constructor(
    private readonly handler: (events: T[]) => void,
    private readonly delay: number,
    private readonly maxSize: number = 100
  ) {}

  push(event: T): void {
    this.buffer.push(event);

    if (this.buffer.length >= this.maxSize) {
      this.flush();
      return;
    }

    if (!this.timeout) {
      this.timeout = setTimeout(() => this.flush(), this.delay);
    }
  }

  flush(): void {
    if (this.timeout) {
      clearTimeout(this.timeout);
      this.timeout = undefined;
    }

    if (this.buffer.length > 0) {
      const events = this.buffer;
      this.buffer = [];
      this.handler(events);
    }
  }

  dispose(): void {
    this.flush();
  }
}
```

### 8.2 이벤트 필터링

```typescript
function filterEvent<T>(
  event: Event<T>,
  predicate: (e: T) => boolean
): Event<T> {
  return (listener, thisArgs?, disposables?) => {
    return event(e => {
      if (predicate(e)) {
        listener.call(thisArgs, e);
      }
    }, null, disposables);
  };
}

// 사용 예시
const significantChanges = filterEvent(
  fileWatcher.onDidChange,
  change => change.size > 1000
);
```

## 9. hdsp-agent 적용 가이드

### 9.1 응답 캐싱

```typescript
class AgentResponseCache {
  private cache = new TTLCache<string, LLMResponse>(5 * 60 * 1000);

  async getOrFetch(
    request: LLMRequest,
    fetcher: () => Promise<LLMResponse>
  ): Promise<LLMResponse> {
    const key = this.createCacheKey(request);
    const cached = this.cache.get(key);

    if (cached) {
      return cached;
    }

    const response = await fetcher();

    // 성공한 응답만 캐싱
    if (!response.error) {
      this.cache.set(key, response);
    }

    return response;
  }

  private createCacheKey(request: LLMRequest): string {
    return JSON.stringify({
      model: request.model,
      messages: request.messages.map(m => ({
        role: m.role,
        content: m.content
      }))
    });
  }
}
```

### 9.2 도구 실행 최적화

```typescript
class OptimizedToolExecutor {
  private readonly sequencer = new Sequencer();
  private readonly cache = new LRUCache<string, ToolResult>(50);

  async execute(tool: Tool, params: object): Promise<ToolResult> {
    // 캐시 가능한 도구 확인
    if (tool.cacheable) {
      const cacheKey = `${tool.id}:${JSON.stringify(params)}`;
      const cached = this.cache.get(cacheKey);
      if (cached) return cached;
    }

    // 순차 실행이 필요한 도구
    if (tool.requiresSequential) {
      return this.sequencer.queue(() => this.executeInternal(tool, params));
    }

    return this.executeInternal(tool, params);
  }

  private async executeInternal(
    tool: Tool,
    params: object
  ): Promise<ToolResult> {
    const result = await tool.execute(params);

    if (tool.cacheable) {
      const cacheKey = `${tool.id}:${JSON.stringify(params)}`;
      this.cache.set(cacheKey, result);
    }

    return result;
  }
}
```
