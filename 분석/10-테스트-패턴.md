### 2.1 TestInstantiationService

```typescript
type ServiceIdCtorPair<T> = [ServiceIdentifier<T>, new (...args: any[]) => T];

class TestInstantiationService extends InstantiationService {
  private readonly _serviceCollection: ServiceCollection;

  constructor(services: ServiceIdCtorPair<any>[] = []) {
    const collection = new ServiceCollection();

    for (const [id, ctor] of services) {
      collection.set(id, new ctor());
    }

    super(collection);
    this._serviceCollection = collection;
  }

  stub<T>(id: ServiceIdentifier<T>, service: T): T {
    this._serviceCollection.set(id, service);
    return service;
  }

  stubPromise<T>(id: ServiceIdentifier<T>, value: T): T {
    return this.stub(id, {
      then: (resolve: (v: T) => void) => resolve(value)
    } as any);
  }
}
```

### 4.3 이벤트 대기

```typescript
async function waitForEvent<T>(
  event: Event<T>,
  timeout: number = 5000
): Promise<T> {
  return new Promise((resolve, reject) => {
    const timer = setTimeout(() => {
      disposable.dispose();
      reject(new Error('Event timeout'));
    }, timeout);

    const disposable = event(value => {
      clearTimeout(timer);
      disposable.dispose();
      resolve(value);
    });
  });
}

// 사용 예시
test('emits event', async () => {
  const emitter = new Emitter<string>();

  setTimeout(() => emitter.fire('hello'), 100);

  const result = await waitForEvent(emitter.event);
  assert.strictEqual(result, 'hello');
});
```

### 4.4 여러 이벤트 수집

```typescript
async function collectEvents<T>(
  event: Event<T>,
  count: number,
  timeout: number = 5000
): Promise<T[]> {
  return new Promise((resolve, reject) => {
    const results: T[] = [];

    const timer = setTimeout(() => {
      disposable.dispose();
      reject(new Error(`Only received ${results.length}/${count} events`));
    }, timeout);

    const disposable = event(value => {
      results.push(value);
      if (results.length === count) {
        clearTimeout(timer);
        disposable.dispose();
        resolve(results);
      }
    });
  });
}
```

## 6. 이벤트 테스트

### 6.1 Emitter 테스트

```typescript
suite('Emitter', () => {
  test('fires event', () => {
    const emitter = new Emitter<number>();
    const values: number[] = [];

    emitter.event(value => values.push(value));

    emitter.fire(1);
    emitter.fire(2);
    emitter.fire(3);

    assert.deepStrictEqual(values, [1, 2, 3]);
  });

  test('dispose removes listeners', () => {
    const emitter = new Emitter<number>();
    let count = 0;

    const disposable = emitter.event(() => count++);

    emitter.fire(1);
    assert.strictEqual(count, 1);

    disposable.dispose();

    emitter.fire(2);
    assert.strictEqual(count, 1); // 변화 없음
  });

  test('dispose emitter', () => {
    const emitter = new Emitter<number>();
    let count = 0;

    emitter.event(() => count++);
    emitter.fire(1);

    emitter.dispose();

    // dispose 후 fire는 무시됨
    emitter.fire(2);
    assert.strictEqual(count, 1);
  });
});
```

## 7. 모킹 패턴

### 7.1 Spy 함수

```typescript
interface Spy<T extends (...args: any[]) => any> {
  (...args: Parameters<T>): ReturnType<T>;
  calls: Parameters<T>[];
  callCount: number;
  lastCall: Parameters<T> | undefined;
  reset(): void;
}

function createSpy<T extends (...args: any[]) => any>(
  implementation?: T
): Spy<T> {
  const calls: Parameters<T>[] = [];

  const spy = function(...args: Parameters<T>): ReturnType<T> {
    calls.push(args);
    return implementation?.(...args);
  } as Spy<T>;

  Object.defineProperty(spy, 'calls', { get: () => calls });
  Object.defineProperty(spy, 'callCount', { get: () => calls.length });
  Object.defineProperty(spy, 'lastCall', { get: () => calls[calls.length - 1] });
  spy.reset = () => calls.length = 0;

  return spy;
}

// 사용 예시
test('calls handler', () => {
  const handler = createSpy((x: number) => x * 2);

  handler(5);
  handler(10);

  assert.strictEqual(handler.callCount, 2);
  assert.deepStrictEqual(handler.calls, [[5], [10]]);
});
```

### 7.2 Stub 서비스

```typescript
class StubFileService implements Partial<IFileService> {
  private files = new Map<string, string>();

  async readFile(uri: URI): Promise<Uint8Array> {
    const content = this.files.get(uri.toString());
    if (!content) {
      throw new Error('File not found');
    }
    return new TextEncoder().encode(content);
  }

  async writeFile(uri: URI, content: Uint8Array): Promise<void> {
    this.files.set(uri.toString(), new TextDecoder().decode(content));
  }

  // 테스트 헬퍼
  setFile(uri: URI, content: string): void {
    this.files.set(uri.toString(), content);
  }

  getFile(uri: URI): string | undefined {
    return this.files.get(uri.toString());
  }
}
```

### 9.1 Assert 확장

```typescript
const assertAsync = {
  async throws(fn: () => Promise<any>, message?: string): Promise<void> {
    try {
      await fn();
      assert.fail(message ?? 'Expected function to throw');
    } catch (e) {
      // 예상된 에러
    }
  },

  async doesNotThrow(fn: () => Promise<any>): Promise<void> {
    try {
      await fn();
    } catch (e) {
      assert.fail(`Unexpected error: ${e}`);
    }
  },

  async eventually(
    fn: () => boolean,
    timeout: number = 5000,
    interval: number = 100
  ): Promise<void> {
    const start = Date.now();
    while (Date.now() - start < timeout) {
      if (fn()) return;
      await sleep(interval);
    }
    assert.fail('Condition not met within timeout');
  }
};
```

```typescript
class TestAgentEnvironment {
```

```typescript
class MockLLMService implements ILLMService {
  private responses: Map<string, LLMResponse> = new Map();

  setResponse(prompt: string, response: LLMResponse): void {
    this.responses.set(prompt, response);
  }

  async sendMessage(request: LLMRequest): Promise<LLMResponse> {
    const key = request.messages[request.messages.length - 1].content;
    const response = this.responses.get(key);

    if (!response) {
      return { content: 'Mock response', toolCalls: [] };
    }

    return response;
  }
}
```
